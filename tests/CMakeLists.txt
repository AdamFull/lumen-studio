function(lumen_add_test name)
    cmake_parse_arguments(ARG "" "" "SOURCES;LIBS" ${ARGN})
    add_executable(${name} ${ARG_SOURCES})
    target_link_libraries(${name} PRIVATE Qt6::Test ${ARG_LIBS})
    lumen_configure_target(${name})
    target_compile_definitions(${name} PRIVATE LUMEN_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()

# Stand-in server for the supervisor/installer tests: no Qt and a static runtime, so a copy runs
# from any folder.
add_executable(fake_server fake_server.cpp)
if(WIN32)
    target_link_libraries(fake_server PRIVATE ws2_32)
endif()
if(MINGW)
    target_link_options(fake_server PRIVATE -static)
endif()
lumen_add_test(tst_runtime SOURCES tst_runtime.cpp MockHttpServer.h LIBS lumen_runtime)
target_compile_definitions(tst_runtime PRIVATE FAKE_SERVER_PATH="$<TARGET_FILE:fake_server>")
find_package(ZLIB QUIET)
if(ZLIB_FOUND)
    target_link_libraries(tst_runtime PRIVATE ZLIB::ZLIB)
else()
    target_link_libraries(tst_runtime PRIVATE z)
endif()
add_dependencies(tst_runtime fake_server)
